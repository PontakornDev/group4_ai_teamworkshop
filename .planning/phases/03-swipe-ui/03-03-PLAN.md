    ```typescript
    import { NextResponse } from "next/server";
    import { auth } from "@/auth";
    import { findUnseenDog, hasUserSeenDog } from "@/lib/storage";

    const RANDOM_DOG_URL = "https://random.dog/woof.json";
    const IMAGE_EXTENSIONS = [".jpg", ".jpeg", ".png", ".gif", ".webp"];
    const MAX_ATTEMPTS = 5;

    function isImageUrl(url: string): boolean {
      return IMAGE_EXTENSIONS.some((ext) => url.toLowerCase().endsWith(ext));
    }

    function extractDogId(url: string): string {
      return url.split("/").pop()?.replace(/\.[^.]+$/, "") ?? "";
    }

    export async function GET(): Promise<NextResponse> {
      // CR-01 (D-07, D-08, D-09): Auth gate — verify session, derive username from it
      const session = await auth();
      if (!session?.user?.name) {
        return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
      }
      const username = session.user.name;

      // Check storage for an unseen dog first (no random.dog call needed)
      const unseen = await findUnseenDog(username);
      if (unseen) {
        return NextResponse.json({ imageUrl: unseen.imageUrl, dogId: unseen.dogId });
      }

      // Fetch a new dog from random.dog — retry up to MAX_ATTEMPTS times
      for (let attempt = 1; attempt <= MAX_ATTEMPTS; attempt++) {
        const res = await fetch(RANDOM_DOG_URL, { cache: "no-store" });
        // CR-02 (D-10): Skip non-2xx responses instead of crashing on res.json()
        if (!res.ok) continue;

        const data = (await res.json()) as { fileSizeBytes: number; url: string };
        if (!data.url || !isImageUrl(data.url)) continue;

        // WR-02 (D-13): Guard against malformed URLs yielding empty dogId
        const dogId = extractDogId(data.url);
        if (!dogId) continue;

        // CR-03 (D-11, D-12): Skip dogs this user has already seen
        if (await hasUserSeenDog(username, dogId)) continue;

        return NextResponse.json({ imageUrl: data.url, dogId });
      }

      return NextResponse.json(
        { error: `Could not fetch a valid dog image after ${MAX_ATTEMPTS} attempts` },
        { status: 500 }
      );
    }
    ```
